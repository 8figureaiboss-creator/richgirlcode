var partsFmt = new Intl.DateTimeFormat("en-US", {

function nyParts(ms) {
  const p = {};
  for (const part of partsFmt.formatToParts(new Date(ms))) {
    if (part.type !== "literal") p[part.type] = Number(part.value);
  }
  return p;
}
function nyOffsetMinutes(ms) {
  const p = nyParts(ms);
  const asUtc = Date.UTC(p.year, p.month - 1, p.day, p.hour, p.minute, p.second);
  return Math.round((asUtc - Math.floor(ms / 1e3) * 1e3) / 6e4);
}
function nyToUtc(date, hhmm) {
  const [y, m, d] = date.split("-").map(Number);
  const clean = hhmm.replace(":", "");
  const h = Number(clean.slice(0, 2));
  const mi = Number(clean.slice(2, 4));
  const naive = Date.UTC(y, m - 1, d, h, mi);
  let t = naive - nyOffsetMinutes(naive) * 6e4;
  const off2 = nyOffsetMinutes(t);
  t = naive - off2 * 6e4;
  return Math.floor(t / 1e3);
}
function nyDate(sec) {
  const p = nyParts(sec * 1e3);
  return `${p.year}-${String(p.month).padStart(2, "0")}-${String(p.day).padStart(2, "0")}`;
}
function addDays(date, n) {
  const [y, m, d] = date.split("-").map(Number);
  const dt = new Date(Date.UTC(y, m - 1, d + n));
  return dt.toISOString().slice(0, 10);
}

var TF_SECONDS = {
  "1M": 60,
  "5M": 300,
  "15M": 900,
  "1H": 3600,
  "4H": 14400,
  "1D": 86400
};
var SOURCE_TF = {
  "1M": "1Min",
  "5M": "5Min",
  "15M": "15Min",
  "1H": "15Min",
  "4H": "15Min",
  "1D": "1Day"
};
var LOOKBACK_DAYS = {
  "1M": 5,
  "5M": 21,
  "15M": 60,
  "1H": 150,
  "4H": 400,
  "1D": 365 * 6
};
function calendarFrom(days) {
  return new Map(days.map((d) => [d.date, d]));
}
function sessionFromAlpaca(row) {
  return {
    date: row.date,
    open: nyToUtc(row.date, row.open),
    close: nyToUtc(row.date, row.close),
    extOpen: nyToUtc(row.date, row.session_open ?? "0400"),
    extClose: nyToUtc(row.date, row.session_close ?? "2000")
  };
}
function bucketOf(t, tf, cal, ext) {
  const day = cal.get(nyDate(t));
  if (!day) return null;
  const start = ext ? day.extOpen : day.open;
  const end = ext ? day.extClose : day.close;
  if (t < start || t >= end) return null;
  if (tf === "1D") return nyToUtc(day.date, "0000");
  const size = TF_SECONDS[tf];
  return start + Math.floor((t - start) / size) * size;
}
function mergeInto(acc, b) {
  const v = acc.v + b.v;
  const pv = (acc.vw ?? acc.c) * acc.v + (b.vw ?? b.c) * b.v;
  return {
    t: acc.t,
    o: acc.o,
    h: Math.max(acc.h, b.h),
    l: Math.min(acc.l, b.l),
    c: b.c,
    v,
    vw: v > 0 ? pv / v : b.vw ?? acc.vw,
    n: (acc.n ?? 0) + (b.n ?? 0)
  };
}
function aggregate(src, tf, cal, ext) {
  const out = [];
  for (const b of src) {
    const bucket = bucketOf(b.t, tf, cal, ext);
    if (bucket === null) continue;
    const last = out[out.length - 1];
    if (last && last.t === bucket) {
      out[out.length - 1] = mergeInto(last, b);
    } else if (!last || bucket > last.t) {
      out.push({ ...b, t: bucket, n: b.n ?? 0 });
    }
  }
  return out;
}
function normalizeDaily(src, cal) {
  const out = [];
  for (const b of src) {
    const date = nyDate(b.t + 3600);
    if (!cal.has(date)) continue;
    out.push({ ...b, t: nyToUtc(date, "0000") });
  }
  return out;
}

async function alpaca(keys, url) {
  const res = await fetch(url, {
    headers: { "APCA-API-KEY-ID": keys.id, "APCA-API-SECRET-KEY": keys.secret, Accept: "application/json" }
  });
  if (res.status === 429) throw new HttpError(429, "rate_limited", "Alpaca rate limit reached (200 calls/min on Basic). Try again in a minute.");
  if (!res.ok) {
    const body = await res.text().catch(() => "");
    throw new HttpError(res.status === 401 || res.status === 403 ? 502 : 502, "upstream", `Alpaca ${res.status}: ${body.slice(0, 200)}`);
  }
  return res.json();
}

async function fetchBars(keys, symbol, timeframe, feed, start, end, maxPages = 8) {
  const out = [];
  let token = "";
  for (let page = 0; page < maxPages; page++) {

      end: new Date(end * 1e3).toISOString(),

      adjustment: "split",
      feed,
      sort: "asc"
    });
    if (token) q.set("page_token", token);
    const json = await alpaca(keys, `${DATA}/v2/stocks/${encodeURIComponent(symbol)}/bars?${q}`);
    for (const b of json.bars ?? []) out.push(toBar(b));
    token = json.next_page_token;

}

function applyTrade(bars, trade, tf, cal, ext) {
  if (!trade) return;
  const bucket = bucketOf(trade.t, tf, cal, ext);
  if (bucket === null) return;
  const last = bars[bars.length - 1];
  if (last && last.t === bucket) {
    last.h = Math.max(last.h, trade.p);
    last.l = Math.min(last.l, trade.p);
    last.c = trade.p;
  } else if (!last || bucket > last.t) {
    bars.push({ t: bucket, o: trade.p, h: trade.p, l: trade.p, c: trade.p, v: 0 });
  }
}
async function handleBars(keys, body) {
  const { symbol, tf, feed, ext } = parseArgs(body);
  const now = Math.floor(Date.now() / 1e3);
  const end = feed === "sip" ? now - SIP_DELAY_SEC : now;
  const startDate = addDays(nyDate(end), -LOOKBACK_DAYS[tf]);
  const days = await sessions(keys, addDays(startDate, -3), addDays(nyDate(end), 1));
  const cal = calendarFrom(days);
  const start = Date.parse(startDate + "T00:00:00Z") / 1e3;
  const raw = await fetchBars(keys, symbol, SOURCE_TF[tf], feed, start, end);
  let bars = tf === "1D" ? normalizeDaily(raw, cal) : aggregate(raw, tf, cal, ext);
  const forming = bucketOf(end, tf, cal, ext);
  if (forming !== null) {
    bars = bars.filter((b) => b.t < forming);
    bars.push(...await minuteTail(keys, symbol, tf, feed, ext, forming, end, cal));
    if (feed === "iex") applyTrade(bars, await latestTrade(keys, symbol), tf, cal, ext);
  }
  return { symbol, tf, ext, bars, formingMinutes: [], provenance: provenance(feed, end), sessions: days };
}

// Bridge between the workspace and Base44: data goes through the otaCharts function,
// saved layouts go into the ChartWorkspace entity. No API keys ever reach this page.
function makeHost() {
  let rowId = null;
  return {
    async call(action, args = {}) {
      try {
        const res = await base44.functions.invoke("otaCharts", { action, ...args });
        return res.data;
      } catch (err) {
        const body = (err && err.response && err.response.data) || {};
        const e = new Error(body.error || err.message || "Request failed");
        e.code = body.code;
        throw e;
      }
    },
    async loadState() {
      const rows = await base44.entities.ChartWorkspace.filter({ key: "main" });
      if (!rows || !rows.length) return null;
      rowId = rows[0].id;
      try {
        return JSON.parse(rows[0].data);
      } catch {
        return null;
      }
    },
    async saveState(state) {
      const data = JSON.stringify(state);
      if (rowId) {
        await base44.entities.ChartWorkspace.update(rowId, { data });
      } else {
        const row = await base44.entities.ChartWorkspace.create({ key: "main", data });
        rowId = row.id;
      }
    },
  };
}

    let app = null;
    let cancelled = false;
    (async () => {
      try {
        const me = await base44.auth.me();
        if (!me || me.role !== "admin") {
          if (!cancelled) setPhase("denied");

      cancelled = true;
      if (app) app.destroy();

        </div>
      )}
      <div ref={hostRef} style={{ height: "100%" }} />
    </div>
  );
}
